\begin{figure}[H]
\centering
\begin{tikzpicture}[
    node distance=9mm and 13mm,
    every node/.style={font=\fontsize{8}{9.5}\selectfont},
    block/.style={rectangle,draw,thin,minimum width=4.55cm,minimum height=8mm,align=center,inner sep=2.2pt},
    decision/.style={diamond,draw,thin,aspect=2.4,align=center,inner sep=1pt},
    arrow/.style={-{Stealth[length=1.8mm]},thin}
]
\node[block] (init) {初始化三视角种群\\(燃油车视角/简化电动车视角/机制视角)};
\node[block,below=of init] (evolve) {三视角混合遗传搜索并行进化\\(选择、有序交叉、教育局部搜索、多样性管理)};
\node[decision,below=9mm of evolve] (budget) {各视角搜索\\预算是否耗尽};
\node[block,below=9mm of budget] (judge) {完整模型裁判\\(补全复算: 车型指派、趟间衔接、非线性充电、\\分时电价与分时碳排、时间窗与站容核验; 各视角留8个精英)};
\node[block,below=of judge] (pool) {精英路线合并去重入池\\(池容量600, 当前最好父方案的路线常驻)};
\node[block,below=of pool] (sp) {路线池上的集合划分重组\\(HiGHS求解, 时间盒不超过总时限的10\%)};
\node[block,below=of sp] (mono) {单调保护接受\\(重组方案严格改善时替换最好父方案, 否则保留原方案)};
\node[block,below=of mono] (output) {输出经独立检查器复核的最好可行方案};
\node[block,right=17mm of judge,minimum width=3.2cm] (dyn) {触发滚动重规划:\\冻结已执行路径前缀,\\继承实体车状态};

\draw[arrow] (init) -- (evolve);
\draw[arrow] (evolve) -- (budget);
\draw[arrow] (budget) -- node[right,xshift=1pt]{是} (judge);
\draw[arrow] (budget.west) -- ++(-11mm,0) node[pos=0,above,xshift=-2mm]{否} |- (evolve.west);
\draw[arrow] (judge) -- (pool);
\draw[arrow] (pool) -- (sp);
\draw[arrow] (sp) -- (mono);
\draw[arrow] (mono) -- (output);
\draw[arrow,dashed] (output.east) -| (dyn.south);
\draw[arrow,dashed] (dyn.north) |- (init.east);
\end{tikzpicture}
\caption{MV-HGS-SP算法流程}
