% improves each fully scheduled feasible solution, then returns seeds/costs.
\node[terminal] (start) at (0,4.15) {开始};
\node[box] (data) at (0,2.9) {输入数据与初始方案\\确定路径搜索成本};
\node[box] (init) at (0,1.5) {个体编码\\生成初始种群};
\node[box,dashed,text width=3.0cm] (dynamic) at (3.65,4.15) {动态事件\\固定已执行作业\\更新客户与车辆状态};
\node[box] (select) at (0,0) {偏差适应度评价\\二元锦标赛选择};
\node[box] (cross) at (0,-1.35) {选择性路径交换};
\node[box] (local) at (0,-2.9) {客户与路径邻域搜索\\多趟次调整、跨中心拆分};
\node[box] (population) at (0,-4.55) {子代入群与概率修复\\罚系数调整、种群管理};
\node[decision] (innerstop) at (0,-6.2) {种群进化\\停止？};

\node[box] (complete) at (7,-6.2) {路径解码与充电排程\\逐一取完整可行方案 $x$};
\node[box] (move) at (7,-4.55) {行程交换、车型置换\\重排充电并评价 $x'$};
\node[decision] (improve) at (7,-2.9) {存在可行改进\\$F(x')<F(x)$？};
\node[smallbox] (accept) at (10.6,-2.9) {接受改进解\\$x\leftarrow x'$};
\node[decision] (all) at (7,-.7) {全部方案均已\\完成搜索？};
\node[box,text width=.34cm,inner sep=2pt] (next) at (4.45,-2.8) {取\\下\\一\\方\\案};
\node[box] (best) at (7,1.15) {更新历轮最优解 $x^*$};
\node[decision] (outerstop) at (7,2.9) {满足终止\\条件？};
\node[terminal] (end) at (7,4.5) {输出 $x^*$};
\node[box,text width=4.8cm] (feedback) at (3.3,-8.15) {更新单位补电成本与趟间充电时间\\以本轮可行方案及 $x^*$ 重建种群};

  node[midway,label,right] {是};

\draw[flow] (outerstop.north) -- (end.south)
  node[midway,label,right] {是};
